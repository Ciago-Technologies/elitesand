//! R5-W1 reproduction: after the log fences, a still-in-flight earlier buffer completes,
//! the log advances and publishes durable_lsn for records whose acks were already
//! reported as FAILED. Clients are told "failed" for records that recovery will replay.
use ciagomem_core::io::{Outcome, SimIo};
use ciagomem_core::slot::CoreControl;
use ciagomem_wal::envelope::KIND_DATA;
use ciagomem_wal::gclog::*;

#[test]
fn r5_defect_w1_failed_ack_later_published_durable() {
    let cfg = LogConfig {
        block: 4096,
        buf_bytes: 16 * 1024,
        buffers: 4,
        max_inflight: 3,
        min_batch_bytes: 8 * 1024,
        max_delay_ns: 5_000,
        capacity: 64 * 1024,
        file: 0,
        tail: TailMode::Pad,
        max_retries: 3,
    };
    let mut io = SimIo::new(1, (cfg.data_offset() + cfg.capacity) as usize);
    let core = CoreControl::default();
    let (mut log, sb) = GroupCommitLog::open(cfg, &Recovery::fresh());
    log.superblock_durable(&sb);
    log.append(1, KIND_DATA, b"a", Some(10), 0, &mut io).unwrap();
    log.poll(1_000_000, true, &mut io);
    log.append(1, KIND_DATA, b"b", Some(11), 0, &mut io).unwrap();
    log.poll(1_000_000, false, &mut io);
    assert_eq!(io.inflight.len(), 2, "two batches in flight");
    let (mut acks, mut failed) = (Vec::new(), Vec::new());
    let mut fenced = false;
    for _ in 0..=cfg.max_retries {
        let i = io.inflight.len() - 1; // batch B (retries are resubmitted at the end)
        let (ud, res) = io.complete(i, Outcome::Err(libc::EIO));
        fenced |= log.on_cqe(ud, res, &core, &mut io, &mut acks, &mut failed).is_err();
    }
    assert!(fenced);
    failed.sort();
    assert_eq!(failed, vec![10, 11], "both acks reported failed");
    // Batch A was healthy and completes after the fence.
    let (ud, res) = io.complete(0, Outcome::Ok);
    let r = log.on_cqe(ud, res, &core, &mut io, &mut acks, &mut failed);
    assert!(r.is_ok(), "on_cqe does not check the fence");
    assert!(core.durable() >= 1, "record of failed ack 10 now published durable");
}
