//! R10: R9-1 retention exercised against a real io_uring ring (Linux only). The SQE is
//! queued while the owner is alive and handed to the kernel only after the owner and
//! the arena/log have been dropped; the kernel then reads/writes retained memory.
#![cfg(target_os = "linux")]
use ciagomem_core::lifecycle::*;
use ciagomem_core::slot::RouteTable;
use ciagomem_core::uring::UringIo;
use ciagomem_core::*;
use std::os::fd::AsRawFd;
fn scratch(name: &str, bytes: usize) -> std::fs::File {
    let p = std::env::temp_dir().join(format!("r10-{name}-{}", std::process::id()));
    let f = std::fs::OpenOptions::new().read(true).write(true).create(true).truncate(true).open(&p).unwrap();
    f.set_len(bytes as u64).unwrap();
    let _ = std::fs::remove_file(&p);
    f
}
#[test]
fn r10_uring_lifecycle_drop_before_kernel_submission_reads_into_retained_arena() {
    let Ok(mut ring) = UringIo::new(8) else { eprintln!("SKIPPED: io_uring unavailable"); return };
    let f = scratch("lc", 8 * 4096);
    ring.register_files(&[f.as_raw_fd()]).unwrap();
    let arena = SlotArena::<512>::new_heap(0, 2, CoreControl::shared(1));
    let route = RouteTable::new(4);
    {
        let cfg = LifecycleConfig { low_free: 0, high_free: 1, max_flush_inflight: 1, max_retries: 2, epoch: 2 };
        let mut lc = SlotLifecycle::new(&arena, &route, 0, 0..2, cfg);
        lc.admit_load(1, 123, RecoveryBounds::clean(2, 1, 5), ExtentLoc::new(0, 2, false), 1, false, &mut ring).unwrap();
    }
    assert!(arena.core(0).is_fenced());
    assert_eq!(arena.io_pending(), 1);
    drop(arena);
    drop(route);
    ring.submit_and_wait(1).unwrap();
    let mut got = Vec::new();
    ring.reap(|ud, res| got.push((ud, res)));
    assert_eq!(got.len(), 1);
    assert_eq!(got[0].1, 4096, "kernel read completed into retained memory");
}
