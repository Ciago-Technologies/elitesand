//! End-to-end on a real kernel: group-commit log over io_uring with O_DIRECT +
//! RWF_DSYNC, registered files and buffers, then crash-recovery scan of the file, and
//! MSG_RING cross-ring wakeups.
#![cfg(target_os = "linux")]

use slae_p0::envelope::KIND_DATA;
use slae_p0::gclog::{GroupCommitLog, LogConfig, Recovery, recover};
use slae_p0::io::{IoBackend, Op, UserData};
use slae_p0::slot::CoreControl;
use slae_p0::uring::UringIo;
use std::fs::OpenOptions;
use std::os::fd::AsRawFd;
use std::os::unix::fs::{FileExt, OpenOptionsExt};

struct Aligned(*mut u8, std::alloc::Layout);
impl Aligned {
    fn new(n: usize) -> Self {
        let l = std::alloc::Layout::from_size_align(n, 4096).unwrap();
        Aligned(unsafe { std::alloc::alloc_zeroed(l) }, l)
    }
}
impl Drop for Aligned {
    fn drop(&mut self) {
        unsafe { std::alloc::dealloc(self.0, self.1) }
    }
}

#[test]
fn group_commit_log_on_real_io_uring_with_o_direct() {
    let path = std::env::temp_dir().join(format!("slae-p0-log-{}", std::process::id()));
    // Prefer the target dir (ext4 here); tmpfs rejects O_DIRECT on some kernels.
    let path = std::path::Path::new(env!("CARGO_TARGET_TMPDIR")).join(path.file_name().unwrap());
    let cfg = LogConfig { capacity: 8 << 20, ..LogConfig::default() };
    let file = OpenOptions::new().read(true).write(true).create(true).truncate(true).custom_flags(libc::O_DIRECT).open(&path).unwrap();
    file.set_len(cfg.data_offset() + cfg.capacity).unwrap();

    let mut io = match UringIo::new(256) {
        Ok(io) => io,
        Err(e) => {
            eprintln!("io_uring unavailable ({e}); skipping");
            return;
        }
    };
    io.register_files(&[file.as_raw_fd()]).unwrap();
    let core = CoreControl::default();
    let (mut log, sb) = GroupCommitLog::open(cfg, &Recovery::fresh());
    for (p, n) in log.buffer_regions() {
        unsafe { io.register_region(p, n).unwrap() };
    }

    // Persist the superblock through the same ring.
    let sbuf = Aligned::new(cfg.block);
    sb.encode(unsafe { std::slice::from_raw_parts_mut(sbuf.0, cfg.block) });
    let ud = UserData { op: Op::Superblock, object: 0, ticket: 1 }.encode();
    unsafe { io.write_at(ud, 0, sbuf.0, cfg.block as u32, sb.file_offset(cfg.block), true).unwrap() };
    io.submit_and_wait(1).unwrap();
    let mut got = Vec::new();
    io.reap(|u, r| got.push((u, r)));
    assert_eq!(got, vec![(ud, cfg.block as i32)]);
    log.superblock_durable(&sb);

    let (mut acks, mut failed) = (Vec::new(), Vec::new());
    let t0 = std::time::Instant::now();
    let n = 20_000u64;
    let mut appended = 0u64;
    while (acks.len() as u64) < n {
        let now = t0.elapsed().as_nanos() as u64;
        while appended < n {
            let payload = appended.to_le_bytes().repeat(4); // 32-byte delta
            match log.append((appended % 977) as u32, KIND_DATA, &payload, Some(appended), now, &mut io) {
                Ok(_) => appended += 1,
                Err(_) => break,
            }
            if appended % 512 == 0 {
                break;
            }
        }
        log.poll(now, appended == n, &mut io);
        io.submit_and_wait(if log.quiescent() { 0 } else { 1 }).unwrap();
        let mut cqes = Vec::new();
        io.reap(|u, r| cqes.push((u, r)));
        for (u, r) in cqes {
            log.on_cqe(u, r, &core, &mut io, &mut acks, &mut failed).unwrap();
        }
    }
    let secs = t0.elapsed().as_secs_f64();
    assert!(failed.is_empty());
    assert!(acks.windows(2).all(|w| w[0] + 1 == w[1]), "acks in order");
    assert_eq!(core.durable(), log.durable_lsn());
    let st = log.stats;
    eprintln!(
        "real io_uring: {n} commits in {:.3}s ({:.0}/s), {} batches ({:.1} rec/batch), WA {:.2} vs 146x baseline",
        secs,
        n as f64 / secs,
        st.batches,
        st.records as f64 / st.batches as f64,
        st.write_amplification()
    );
    assert!(st.write_amplification() < 4.0);

    // Recovery scan through O_DIRECT reads.
    let rfile = OpenOptions::new().read(true).custom_flags(libc::O_DIRECT).open(&path).unwrap();
    let mut seen = 0u64;
    let rec = recover(
        &cfg,
        &mut |off, buf| {
            let a = Aligned::new(buf.len());
            let s = unsafe { std::slice::from_raw_parts_mut(a.0, buf.len()) };
            rfile.read_exact_at(s, off)?;
            buf.copy_from_slice(s);
            Ok(())
        },
        &mut |h, p| {
            assert_eq!(p, &seen.to_le_bytes().repeat(4)[..], "lsn {}", h.lsn);
            seen += 1;
        },
    )
    .unwrap();
    assert_eq!(seen, n);
    assert_eq!(rec.records, n);
    drop(io);
    std::fs::remove_file(&path).ok();
}

#[test]
fn msg_ring_wakes_another_ring() {
    let (Ok(mut a), Ok(mut b)) = (UringIo::new(8), UringIo::new(8)) else {
        eprintln!("io_uring unavailable; skipping");
        return;
    };
    let wake = UserData { op: Op::Wake, object: 3, ticket: 42 }.encode();
    a.post_wake(b.ring_fd(), wake).unwrap();
    a.submit().unwrap();
    b.submit_and_wait(1).unwrap();
    let mut got = Vec::new();
    b.reap(|u, r| got.push((u, r)));
    assert_eq!(got.len(), 1);
    assert_eq!(UserData::decode(got[0].0), Some(UserData { op: Op::Wake, object: 3, ticket: 42 }));
    // The sender's CQE was suppressed (SKIP_SUCCESS).
    let mut n = 0;
    a.reap(|_, _| n += 1);
    assert_eq!(n, 0);
}
